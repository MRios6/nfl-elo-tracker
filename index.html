const K_OVERALL = 20; const K_STAT = 15; const HFA = 48; // Home Field Advantage let teamData = {}; function resetTeamData() { teamData = {}; } function initTeam(abbr) { if (!teamData[abbr]) { teamData[abbr] = { overall: 1500, off: 1500, def: 1500, home: 1500, away: 1500 }; } } async function fetchFullSeasonGames() { const statusEl = document.getElementById('status'); statusEl.innerText = "Processing full season (Weeks 1 to 18)..."; // Reset Elos to 1500 base before recalculating full history resetTeamData(); const TOTAL_WEEKS = 18; let totalProcessed = 0; try { for (let week = 1; week <= TOTAL_WEEKS; week++) { statusEl.innerText = `Fetching Week ${week} of 18...`; // Query ESPN API for each specific week const response = await fetch(`https://site.api.espn.com/apis/site/v2/sports/football/nfl/scoreboard?seasontype=2&week=${week}`); const data = await response.json(); if (data.events) { data.events.forEach(event => { const comp = event.competitions[0]; // Only process games that are finished if (comp.status.type.completed) { const home = comp.competitors.find(c => c.homeAway === 'home'); const away = comp.competitors.find(c => c.homeAway === 'away'); processGame(home, away); totalProcessed++; } }); } } // Save fresh season calculations to LocalStorage localStorage.setItem('nfl_elo_data', JSON.stringify(teamData)); renderTable(); populateDropdowns(); statusEl.innerText = `Done! Processed ${totalProcessed} total season games.`; } catch (err) { statusEl.innerText = "Error fetching full season data."; console.error(err); } } function processGame(home, away) { const hAbbr = home.team.abbreviation; const aAbbr = away.team.abbreviation; initTeam(hAbbr); initTeam(aAbbr); const hScore = parseInt(home.score); const aScore = parseInt(away.score); // Extract yards safely from statistics array if available const hYards = parseInt(home.statistics?.find(s => s.name === 'totalYards')?.displayValue || 300); const aYards = parseInt(away.statistics?.find(s => s.name === 'totalYards')?.displayValue || 300); // 1. Overall Elo Calculation const hElo = teamData[hAbbr].overall; const aElo = teamData[aAbbr].overall; const eHome = 1 / (1 + Math.pow(10, (aElo - (hElo + HFA)) / 400)); const actualHome = hScore > aScore ? 1 : (hScore === aScore ? 0.5 : 0); const mov = Math.abs(hScore - aScore); const movMult = Math.log(mov + 1) * (2.2 / ((actualHome === 1 ? hElo - aElo : aElo - hElo) * 0.001 + 2.2)); teamData[hAbbr].overall = Math.round(hElo + K_OVERALL * movMult * (actualHome - eHome)); teamData[aAbbr].overall = Math.round(aElo + K_OVERALL * movMult * ((1 - actualHome) - (1 - eHome))); // 2. Offense & Defense Rating Adjustment const yardDiff = hYards - aYards; teamData[hAbbr].off += Math.round(K_STAT * (yardDiff / 100)); teamData[aAbbr].def -= Math.round(K_STAT * (yardDiff / 100)); // 3. Home / Away Split Adjustment if (actualHome === 1) { teamData[hAbbr].home += 10; teamData[aAbbr].away -= 10; } else if (actualHome === 0) { teamData[hAbbr].home -= 10; teamData[aAbbr].away += 10; } } function calculatePrediction() { const home = document.getElementById('homeSelect').value; const away = document.getElementById('awaySelect').value; const resBox = document.getElementById('predictionResult'); if (!home || !away || home === away) { resBox.innerText = "Select two different teams."; return; } const hOff = teamData[home].off; const aDef = teamData[away].def; const aOff = teamData[away].off; const hDef = teamData[home].def; const eloDiff = (hOff - aDef) + (hDef - aOff) + HFA; const winProb = (1 / (1 + Math.pow(10, -eloDiff / 400)) * 100).toFixed(1); const spread = (eloDiff / 25).toFixed(1); let favorite = eloDiff > 0 ? home : away; let absSpread = Math.abs(spread); resBox.innerHTML = ` ${home} Win Probability: ${winProb}%
Projected Spread: ${favorite} -${absSpread} `; } function populateDropdowns() { const hSelect = document.getElementById('homeSelect'); const aSelect = document.getElementById('awaySelect'); const teams = Object.keys(teamData).sort(); hSelect.innerHTML = ''; aSelect.innerHTML = ''; teams.forEach(t => { hSelect.innerHTML += ``; aSelect.innerHTML += ``; }); } function renderTable() { const tbody = document.getElementById('eloBody'); tbody.innerHTML = ''; const sortedTeams = Object.keys(teamData).sort((a,b) => teamData[b].overall - teamData[a].overall); sortedTeams.forEach(team => { const d = teamData[team]; tbody.innerHTML += ` ${team} ${d.overall} ${d.off} ${d.def} ${d.home} ${d.away} `; }); } // Update button event listener to trigger full season sync document.querySelector('button').setAttribute('onclick', 'fetchFullSeasonGames()'); // Load stored data on boot if available let stored = localStorage.getItem('nfl_elo_data'); if (stored) { teamData = JSON.parse(stored); renderTable(); populateDropdowns(); }
